local DefaultSettings = { AntiAdmin = true }

local function LoadConfig()
    if isfolder and not isfolder(ConfigFolder) then pcall(function() makefolder(ConfigFolder) end) end
    if isfile and isfile(ConfigFile) then
        local success, result = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)

        pcall(function()

        end)
    end
end

-- Server Hop & Anti-Admin
local isHopping = false
local function ServerHop()
    if isHopping then return end
    isHopping = true
    task.spawn(function()
        pcall(function()
            local events = ReplicatedStorage:FindFirstChild("Events")
            if events and events:FindFirstChild("GetSettings") then
                events.GetSettings:InvokeServer()
            end
        end)
        task.wait(0.5)
        
        local serversApi = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local req = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
        if req then
            local success, response = pcall(function() return req({Url = serversApi, Method = "GET"}) end)
            if success and response.StatusCode == 200 then
                local data = HttpService:JSONDecode(response.Body)
                if data and data.data then
                    for _, server in ipairs(data.data) do
                        if server.playing < server.maxPlayers and server.id ~= game.JobId then
                            TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LocalPlayer)
                            task.wait(5)
                            return
                        end
                    end
                end
            end
        end
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    end)
end

local function CheckPlayerForAdmin(player)
    if player == LocalPlayer then return end
    task.spawn(function()
        local success, rank = pcall(function() return player:GetRankInGroup(GROUP_ID) end)
        if success and rank and rank > 1 then
            if getgenv().AntiAdminEnabled then
                local msg = "**Admin Detected**: `" .. player.Name .. "`\nServer Hopping..."
                sendNotification("Anti-Admin", "Admin: " .. player.Name .. "\nServer Hopping...", 5)
                safeSendWebhook(msg)
                ServerHop()
            end
        end
    end)
end

-- Crafting & UI Bypass Helpers
local function triggerPromptByKeyword(keywords)
    for _, prompt in ipairs(workspace:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") then
            local pName = string.lower(prompt.Name)
            local pParent = string.lower(prompt.Parent and prompt.Parent.Name or "")
            local pAction = string.lower(prompt.ActionText or "")
            local pObject = string.lower(prompt.ObjectText or "")
            
            for _, key in ipairs(keywords) do
                if string.find(pName, key) or string.find(pParent, key) or string.find(pAction, key) or string.find(pObject, key) then
                    prompt.RequiresLineOfSight = false
                    prompt.MaxActivationDistance = 999999
                    pcall(function() fireproximityprompt(prompt) end)
                    break
                end
            end
        end
    end
end

local function forceOpenUIByKeyword(keywords)
    for _, gui in ipairs(PlayerGui:GetChildren()) do
        if gui:IsA("ScreenGui") then
            local gName = string.lower(gui.Name)
            for _, key in ipairs(keywords) do
                if string.find(gName, key) then
                    gui.Enabled = true
                    for _, frame in ipairs(gui:GetDescendants()) do
                        if frame:IsA("Frame") or frame:IsA("CanvasGroup") then frame.Visible = true end
                    end
                    break
                end
            end
        end
    end
end

-- Equipment Helpers
local function EquipPickaxe()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if not char or not backpack then return end

    local equippedTool = char:FindFirstChildOfClass("Tool")
    if not equippedTool or (not string.find(string.lower(equippedTool.Name), "pick") and not string.find(string.lower(equippedTool.Name), "mine")) then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and (string.find(string.lower(item.Name), "pick") or string.find(string.lower(item.Name), "mine") or string.find(string.lower(item.Name), "axe")) then
                char.Humanoid:EquipTool(item)
                break
            end
        end
    end
end

end

-- Ore Logic & QTE
local LastTransparentOre = nil
local function HandleOreTransparency(currentOre)
    if LastTransparentOre ~= currentOre then
        if LastTransparentOre and LastTransparentOre.Parent then
            for _, part in ipairs(LastTransparentOre:GetDescendants()) do
                if part:IsA("BasePart") then part.LocalTransparencyModifier = 0 end
            end
            if LastTransparentOre:IsA("BasePart") then LastTransparentOre.LocalTransparencyModifier = 0 end
        end
        LastTransparentOre = currentOre
        if currentOre then
            for _, part in ipairs(currentOre:GetDescendants()) do
                if part:IsA("BasePart") then part.LocalTransparencyModifier = 0.85 end
            end
            if currentOre:IsA("BasePart") then currentOre.LocalTransparencyModifier = 0.85 end
        end
    end
end

local function getQTEMiningFrames()
    local char = LocalPlayer.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        local miningGui = char.HumanoidRootPart:FindFirstChild("Mining")
        if miningGui and miningGui:IsA("BillboardGui") then
            local uiFrame = miningGui:FindFirstChild("Frame")
            if uiFrame and uiFrame.Visible and uiFrame:FindFirstChild("Amount") then return uiFrame.Amount, uiFrame end
        end
    end
    
    local qte = PlayerGui:FindFirstChild("QuickTimeEvents")
    if qte then
        local events = qte:FindFirstChild("QTEClient") and qte.QTEClient:FindFirstChild("Events")
        if events then
            local miningEvent = events:FindFirstChild("Mining")
            if miningEvent then
                local miningInner = miningEvent:FindFirstChild("Mining")
                if miningInner and miningInner.Enabled then
                    local uiFrame = miningInner:FindFirstChild("Frame")
                    if uiFrame and uiFrame.Visible and uiFrame:FindFirstChild("Amount") then return uiFrame.Amount, uiFrame end
                end
            end
        end
    end
    return nil, nil
end

local function getTargetOresList()
    local safeOres = {}
    local islands = workspace:FindFirstChild("Islands")
    local selectedZone = getgenv().TargetZone or "Todos"
    if not islands then return safeOres end
    
    local function scanIslandForOres(islandName)
        local islandFolder = islands:FindFirstChild(islandName)
        if islandFolder and islandFolder:FindFirstChild("Island") then
            for _, obj in pairs(islandFolder.Island:GetChildren()) do
                if string.match(obj.Name, "^OreBlock") and obj:GetAttribute("Mined") ~= true then
                    table.insert(safeOres, obj)
                end
            end
        end
    end

    if selectedZone == "Todos" or selectedZone == "Anchor Town" then scanIslandForOres("Anchor Town") end
    if selectedZone == "Todos" or selectedZone == "Maple Village" then scanIslandForOres("Maple Village") end
    if selectedZone == "Todos" or selectedZone == "Clown Town" then scanIslandForOres("Clown Town") end
    return safeOres
end

local function findNearestOre()
    local nearestOre, shortestDistance = nil, math.huge
    local char = LocalPlayer.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        local myPos = char.HumanoidRootPart.Position
        for _, obj in pairs(getTargetOresList()) do
            local orePos = nil
            if obj:IsA("Model") and obj.PrimaryPart then
                orePos = obj.PrimaryPart.Position
            elseif obj:IsA("BasePart") then
                orePos = obj.Position
            else
                local part = obj:FindFirstChildWhichIsA("BasePart")
                if part then orePos = part.Position end
            end
            if orePos then
                local distance = (orePos - myPos).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestOre = obj
                end
            end
        end
    end
    return nearestOre, shortestDistance
end

local function CalculateRelease(amountBar, parentFrame)
    if parentFrame.AbsoluteSize.X > 5 and parentFrame.AbsoluteSize.Y > 5 then
        local isHorizontal = parentFrame.AbsoluteSize.X > parentFrame.AbsoluteSize.Y
        local targetZone = nil
        for _, child in ipairs(parentFrame:GetChildren()) do
            if (child:IsA("Frame") or child:IsA("ImageLabel")) and child.Name ~= "Amount" and child.Name ~= "Background" and child.Name ~= "UIStroke" and child.Name ~= "UICorner" then
                if child:IsA("Frame") and child.BackgroundColor3.G > 0.4 then
                    targetZone = child
                    break
                elseif not targetZone then
                    targetZone = child
                end
            end
        end
        local requiredFill = FALLBACK_TARGET
        if targetZone then
            if isHorizontal then
                local parentWidth = parentFrame.AbsoluteSize.X
                local targetHitX = targetZone.AbsolutePosition.X + (targetZone.AbsoluteSize.X * 0.80)
                local parentLeft = parentFrame.AbsolutePosition.X
                requiredFill = (targetHitX - parentLeft) / parentWidth
            else
                local parentHeight = parentFrame.AbsoluteSize.Y
                local targetHitY = targetZone.AbsolutePosition.Y + (targetZone.AbsoluteSize.Y * 0.20)
                local parentBottom = parentFrame.AbsolutePosition.Y + parentFrame.AbsoluteSize.Y
                requiredFill = (parentBottom - targetHitY) / parentHeight
            end
        end
        local currentFill = isHorizontal and (amountBar.AbsoluteSize.X / parentFrame.AbsoluteSize.X) or (amountBar.AbsoluteSize.Y / parentFrame.AbsoluteSize.Y)
        if currentFill >= (requiredFill - LAG_OFFSET) then
            return true
        end
    end
    return false
end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and string.find(string.lower(obj.Name), "smuggler") then
            found = true
            for _, v in ipairs(obj:GetDescendants()) do
                if v:IsA("StringValue") or v:IsA("Tool") or v:IsA("Model") then table.insert(items, v.Name) end
            end
        end
    end
    if found then
        sendNotification("Smuggler", #items > 0 and ("Items: " .. table.concat(items, ", ")) or "Smuggler NPC found!", 5)
    else
        sendNotification("Smuggler", "Smuggler NPC not found!", 3)
    end
end)

TrackerLayout.Padding = UDim.new(0, 3)

local function GetPlayerStats(player)
    local level, bounty = "N/A", "N/A"
    local leaderstats = player:FindFirstChild("leaderstats")
    if leaderstats then
        local lvlObj = leaderstats:FindFirstChild("Level") or leaderstats:FindFirstChild("Lvl")
        if lvlObj then level = tostring(lvlObj.Value) end
        local bntObj = leaderstats:FindFirstChild("Bounty") or leaderstats:FindFirstChild("Belly") or leaderstats:FindFirstChild("Berries")
        if bntObj then bounty = tostring(bntObj.Value) end
    end
    return level, bounty
end

local function UpdatePlayerTracker()
    for _, child in pairs(TrackerScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end

    local myChar = LocalPlayer.Character
    local myPos = (myChar and myChar:FindFirstChild("HumanoidRootPart")) and myChar.HumanoidRootPart.Position or Vector3.zero

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local lvl, bnt = GetPlayerStats(plr)
            local distStr = "?"
            local pChar = plr.Character
            if pChar and pChar:FindFirstChild("HumanoidRootPart") and myPos ~= Vector3.zero then
                distStr = tostring(math.floor((pChar.HumanoidRootPart.Position - myPos).Magnitude)) .. "m"
            end

            local pBtn = Instance.new("TextButton", TrackerScroll)
            pBtn.Size = UDim2.new(1, -2, 0, 20)
            pBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
            pBtn.TextColor3 = Color3.fromRGB(220, 220, 220)
            pBtn.Font = Enum.Font.Gotham
            pBtn.TextSize = 9
            pBtn.TextXAlignment = Enum.TextXAlignment.Left
            pBtn.Text = string.format(" ID:%d | Lvl:%s | Bnt:%s | %s", plr.UserId, lvl, bnt, distStr)

            local pCorner = Instance.new("UICorner", pBtn)
            pCorner.CornerRadius = UDim.new(0, 4)

            pBtn.MouseButton1Click:Connect(function()
                if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                    TeleportToPos(plr.Character.HumanoidRootPart.Position + Vector3.new(0, 3, 0), "ID " .. tostring(plr.UserId))
                end
            end)
        end
    end
end

task.spawn(function()
    while getgenv().AutoFarmingRunning do
        pcall(UpdatePlayerTracker)
        task.wait(3)
    end
end)

-- ==========================================
-- MAIN WORKER LOOP
-- ==========================================
task.spawn(function()
    local lastCastTime = 0
    while true do
        task.wait(0.01)

        -- 1. UNDERGROUND AUTO MINE
        if getgenv().AutoMineEnabled then
            EquipPickaxe()
            local nearestOre, distance = findNearestOre()
            local char = LocalPlayer.Character
            
            if nearestOre and char and char:FindFirstChild("HumanoidRootPart") then
                local targetPos = nil
                if nearestOre:IsA("Model") and nearestOre.PrimaryPart then
                    targetPos = nearestOre.PrimaryPart.Position
                elseif nearestOre:IsA("BasePart") then
                    targetPos = nearestOre.Position
                else
                    local p = nearestOre:FindFirstChildWhichIsA("BasePart")
                    if p then targetPos = p.Position end
                end

                if targetPos then
                    local undergroundPos = targetPos - Vector3.new(0, 3.0, 0)
                    char:PivotTo(CFrame.lookAt(undergroundPos, targetPos))
                    HandleOreTransparency(nearestOre)

                    local amountBar, parentFrame = getQTEMiningFrames()
                    if amountBar and parentFrame then
                        if CalculateRelease(amountBar, parentFrame) then
                            MouseController:Set(false)
                            task.wait(0.03)
                        else
                            MouseController:Set(true)
                        end
                    else
                        MouseController:Set(true)
                        task.wait(0.08)
                        MouseController:Set(false)
                        task.wait(0.08)
                    end
                end
            else
                HandleOreTransparency(nil)
                MouseController:Set(false)
            end
            continue
        end

        -- 2. HYBRID REMOTE AUTO FISH
        if getgenv().AutoFishEnabled then
            if isMinigameActive then continue end

            EquipFishingRod()

            local qte = PlayerGui:FindFirstChild("QuickTimeEvents")
            local shakeActive = false
            if qte then
                local btn = qte:FindFirstChild("Button", true) or qte:FindFirstChild("Shake", true)
                if btn and (btn.Visible or (btn:IsA("GuiObject") and btn.AbsoluteSize.X > 0)) then
                    shakeActive = true
                    MouseController:Set(false)
                    local inset = GuiService:GetGuiInset()
                    local btnX = btn.AbsolutePosition.X + (btn.AbsoluteSize.X / 2)
                    local btnY = btn.AbsolutePosition.Y + (btn.AbsoluteSize.Y / 2) + inset.Y
                    VirtualInputManager:SendMouseButtonEvent(btnX, btnY, 0, true, game, 1)
                    task.wait(0.01)
                    VirtualInputManager:SendMouseButtonEvent(btnX, btnY, 0, false, game, 1)
                    task.wait(0.05)
                end
            end

            if shakeActive then continue end

            if tick() - lastCastTime > 1.8 then
                TriggerPerfectCast()
                lastCastTime = tick()
            end
        else
            if MouseController.IsDown then MouseController:Set(false) end
            task.wait(0.2)
        end
    end
end)
